"""
Module 036 - DFA Validator
"""

import numpy as np

from .engine import DFAResult
from .statistics import DFAStatistics


class DFAValidator:

    def validate(
        self,
        result: DFAResult,
        statistics: DFAStatistics,
    ) -> bool:

        issues = []

        if result.sample_size < 200:
            issues.append(f"Sample size too small: {result.sample_size}")

        # Allow tolerance for finite-sample estimation error
        if not (-0.2 < result.alpha < 1.3):
            issues.append(
                f"Alpha out of valid range: {result.alpha:.4f} "
                f"(expected -0.2 to 1.3)"
            )

        if len(result.scales) < 5:
            issues.append(
                f"Too few windows: {len(result.scales)} (minimum 5)"
            )

        if len(result.scales) != len(result.fluctuation):
            issues.append("Scales and fluctuation length mismatch")

        if np.isnan(result.fluctuation).any():
            issues.append("Fluctuation values contain NaN")

        if np.isinf(result.fluctuation).any():
            issues.append("Fluctuation values contain Inf")

        if not np.isfinite(result.slope):
            issues.append(f"Invalid regression slope: {result.slope}")

        if not np.isfinite(result.intercept):
            issues.append(f"Invalid regression intercept: {result.intercept}")

        if not np.isclose(statistics.alpha, result.alpha, atol=1e-10):
            issues.append(
                f"Alpha mismatch: stats={statistics.alpha:.6f}, "
                f"result={result.alpha:.6f}"
            )

        if statistics.r_squared < 0.80:
            issues.append(
                f"Low R-squared: {statistics.r_squared:.4f} (warning threshold: 0.80)"
            )

        if not result.series_type or result.series_type == "unknown":
            issues.append("Series type not specified")

        if statistics.series_type != result.series_type:
            issues.append("Series type mismatch")

        if issues:
            raise ValueError(
                f"DFA validation failed with {len(issues)} issue(s):\n" +
                "\n".join(f"  - {issue}" for issue in issues)
            )

        return True


def run_validation(result, statistics):
    """Run validation and print results."""
    validator = DFAValidator()

    print("\n" + "=" * 60)
    print("DFA VALIDATION")
    print("=" * 60)

    try:
        validator.validate(result, statistics)
        print("\n  ✓ All validation checks passed")
        print("=" * 60)
        return True
    except ValueError as e:
        print(e)
        print("=" * 60)
        return False
