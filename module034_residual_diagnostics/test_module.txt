def test_white_noise_residuals():
    """Test 1: White noise residuals should pass all tests."""
    print("\n" + "=" * 60)
    print("TEST 1: White Noise Residuals")
    print("=" * 60)

    engine = ResidualDiagnosticsEngine(lb_lag=20)
    stats_engine = ResidualDiagnosticsStatisticsEngine()

    np.random.seed(42)
    n = 2000
    white_noise = np.random.normal(0, 1, n)
    result = engine.compute(white_noise)
    stats = stats_engine.analyze(result)

    assert result.sample_size == n
    assert abs(result.residual_mean) < 0.1
    assert 0.5 < result.residual_variance < 2.0
    assert abs(result.residual_skewness) < 0.3
    assert abs(result.residual_kurtosis) < 0.5

    # White noise should pass all tests
    assert not stats.serial_correlation_rejected, "White noise should not have serial correlation"
    assert not stats.arch_remaining_rejected, "White noise should not have ARCH effects"

    print(f"  Skewness: {result.residual_skewness:.4f}")
    print(f"  Kurtosis: {result.residual_kurtosis:.4f}")
    print(f"  JB p-value: {result.jb_pvalue:.6f}")
    print(f"  All tests passed: {stats.all_tests_passed}")
    print("  ✓ White noise test passed")
    return True


def test_fat_tailed_residuals():
    """Test 2: Fat-tailed residuals should fail normality but pass serial tests."""
    print("\n" + "=" * 60)
    print("TEST 2: Fat-Tailed Residuals")
    print("=" * 60)

    engine = ResidualDiagnosticsEngine(lb_lag=20)
    stats_engine = ResidualDiagnosticsStatisticsEngine()

    np.random.seed(42)
    n = 2000
    fat_tailed = np.random.standard_t(df=3, size=n)
    result = engine.compute(fat_tailed)
    stats = stats_engine.analyze(result)

    # Should fail normality
    assert stats.normality_rejected, "Fat-tailed should fail normality test"

    # Should pass serial correlation (independent draws)
    assert not stats.serial_correlation_rejected, "Independent draws should not have serial correlation"

    print(f"  Kurtosis: {result.residual_kurtosis:.4f}")
    print(f"  JB p-value: {result.jb_pvalue:.6f}")
    print(f"  Normality rejected: {stats.normality_rejected}")
    print(f"  Serial corr rejected: {stats.serial_correlation_rejected}")
    print("  ✓ Fat-tailed test passed")
    return True


def test_arch_residuals():
    """Test 3: ARCH residuals should fail LB squared test."""
    print("\n" + "=" * 60)
    print("TEST 3: ARCH Residuals")
    print("=" * 60)

    engine = ResidualDiagnosticsEngine(lb_lag=20)
    stats_engine = ResidualDiagnosticsStatisticsEngine()

    np.random.seed(42)
    n = 2000

    omega, alpha = 0.1, 0.7
    returns = np.zeros(n)
    sigma2 = np.ones(n) * omega
    for t in range(1, n):
        sigma2[t] = omega + alpha * returns[t-1]**2
        returns[t] = np.sqrt(sigma2[t]) * np.random.normal(0, 1)

    result = engine.compute(returns)
    stats = stats_engine.analyze(result)

    assert stats.arch_remaining_rejected, "ARCH process should fail LB squared test"

    print(f"  LB squared p-value: {result.lb_squared_pvalue:.6f}")
    print(f"  ARCH remaining: {stats.arch_remaining_rejected}")
    print("  ✓ ARCH residuals test passed")
    return True
