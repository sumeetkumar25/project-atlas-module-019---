def test_random_walk_returns():
    """Test 1: Random walk returns should give H ≈ 0.5."""
    print("\n" + "=" * 60)
    print("TEST 1: Random Walk Returns")
    print("=" * 60)

    engine = HurstEngine(min_window=16, num_windows=20)
    stats_engine = HurstStatisticsEngine()

    np.random.seed(42)
    n = 2000
    returns = np.random.normal(0, 1, n)
    result = engine.compute(returns, series_type="log_returns")
    stats = stats_engine.analyze(result)

    assert 0.40 < result.hurst_exponent < 0.60, (
        f"Random walk returns H should be ~0.5, got {result.hurst_exponent:.4f}"
    )
    assert stats.series_type == "log_returns"
    assert stats.r_squared > 0.85, f"R-squared should be high, got {stats.r_squared:.4f}"

    print(f"  Hurst: {result.hurst_exponent:.4f}")
    print(f"  R²: {stats.r_squared:.4f}")
    print(f"  Series type: {stats.series_type}")
    print(f"  Regime: {stats.regime}")
    print("  ✓ Random walk test passed")
    return True


def test_persistent_returns():
    """Test 2: Persistent returns should give H > 0.5."""
    print("\n" + "=" * 60)
    print("TEST 2: Persistent Returns")
    print("=" * 60)

    engine = HurstEngine(min_window=16, num_windows=20)
    stats_engine = HurstStatisticsEngine()

    np.random.seed(42)
    n = 2000

    phi = 0.3
    persistent = np.zeros(n)
    persistent[0] = np.random.normal(0, 1)
    for t in range(1, n):
        persistent[t] = phi * persistent[t-1] + np.random.normal(0, 1)

    result = engine.compute(persistent, series_type="persistent_returns")
    stats = stats_engine.analyze(result)

    assert result.hurst_exponent > 0.50, (
        f"Persistent returns H should be > 0.50, got {result.hurst_exponent:.4f}"
    )
    assert stats.is_persistent or stats.is_random_walk

    print(f"  Hurst: {result.hurst_exponent:.4f}")
    print(f"  Regime: {stats.regime}")
    print("  ✓ Persistent test passed")
    return True


def test_mean_reverting_returns():
    """Test 3: Mean-reverting returns should give H < 0.5."""
    print("\n" + "=" * 60)
    print("TEST 3: Mean-Reverting Returns")
    print("=" * 60)

    engine = HurstEngine(min_window=16, num_windows=20)
    stats_engine = HurstStatisticsEngine()

    np.random.seed(42)
    n = 2000

    # Mean-reverting: negative autocorrelation
    mr = np.zeros(n)
    mr[0] = np.random.normal(0, 1)
    for t in range(1, n):
        mr[t] = -0.3 * mr[t-1] + np.random.normal(0, 1)

    result = engine.compute(mr, series_type="mean_reverting_returns")
    stats = stats_engine.analyze(result)

    assert result.hurst_exponent < 0.55, (
        f"Mean-reverting returns H should be < 0.55, got {result.hurst_exponent:.4f}"
    )
    assert stats.is_mean_reverting or stats.is_random_walk

    print(f"  Hurst: {result.hurst_exponent:.4f}")
    print(f"  Regime: {stats.regime}")
    print("  ✓ Mean-reverting test passed")
    return True

def test_pipeline():
    """Test 6: Full pipeline."""
    print("\n" + "=" * 60)
    print("TEST 6: Full Pipeline")
    print("=" * 60)

    n = 500
    np.random.seed(42)
    close = np.cumsum(np.random.normal(0, 1, n)) + 100
    close = np.maximum(close, 1)

    dates = pd.date_range(start="2020-01-01", periods=n, freq="15min")
    df = pd.DataFrame({
        "timestamp": dates,
        "open": close * 0.999,
        "high": close * 1.002,
        "low": close * 0.998,
        "close": close,
        "volume": np.ones(n) * 100,
    })

    research = HurstResearch()
    result = research.run_dataframe(df, dataset_id="pipeline_test")

    assert result.status == "completed"
    assert result.statistics.get("series_type") == "log_returns"
    assert "hurst_statistics" in result.artifacts
    assert "hurst_results" in result.artifacts
    assert "hurst_report" in result.artifacts
    assert result.statistics.get("validation_passed") == True

    print(f"  Series type: {result.statistics.get('series_type')}")
    print(f"  Hurst: {result.statistics.get('hurst_exponent'):.4f}")
    print("  ✓ Full pipeline test passed")
    return True
