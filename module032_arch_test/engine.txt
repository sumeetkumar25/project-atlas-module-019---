"""
Module 032 - ARCH Engine

Computes ARCH LM test for volatility clustering in BTC returns.

Uses statsmodels het_arch for correct statistical implementation.
"""

from dataclasses import dataclass, field

import numpy as np
import pandas as pd
from statsmodels.stats.diagnostic import het_arch


# ----------------------------------------------------
# Result Container
# ----------------------------------------------------

@dataclass
class ARCHResult:
    lags: np.ndarray
    lm_statistics: np.ndarray
    p_values: np.ndarray

    sample_size: int
    alpha: float
    max_lag: int

    confidence_level: float = 0.0
    significant_mask: np.ndarray = field(default=None)

    def __post_init__(self):
        self.confidence_level = 1.0 - self.alpha
        if self.significant_mask is None:
            self.significant_mask = self.p_values < self.alpha


# ----------------------------------------------------
# ARCH Engine
# ----------------------------------------------------

class ARCHEngine:

    def __init__(self, max_lag: int = 20, alpha: float = 0.05):
        self.max_lag = max_lag
        self.alpha = alpha

    @staticmethod
    def compute_log_returns(close: pd.Series) -> pd.Series:
        returns = np.log(close / close.shift(1))
        return returns.dropna()

    def compute(self, returns: pd.Series) -> ARCHResult:
        # Demean returns
        resid = returns - returns.mean()

        # ARCH LM test for each lag
        lm_stats = np.zeros(self.max_lag)
        p_vals = np.zeros(self.max_lag)

        for lag in range(1, self.max_lag + 1):
            result = het_arch(resid, nlags=lag)
            lm_stats[lag - 1] = result[0]
            p_vals[lag - 1] = result[1]

        return ARCHResult(
            lags=np.arange(1, self.max_lag + 1),
            lm_statistics=lm_stats,
            p_values=p_vals,
            sample_size=len(returns),
            alpha=self.alpha,
            max_lag=self.max_lag,
        )
