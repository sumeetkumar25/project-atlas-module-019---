"""
Module 032 - ARCH Statistics

Computes descriptive statistics from ARCH test results.
"""

from dataclasses import dataclass, asdict
from typing import Optional

import numpy as np
import pandas as pd

from .engine import ARCHResult


@dataclass
class ARCHStatistics:
    sample_size: int
    alpha: float
    max_lag: int

    significant_lags: int

    first_significant_lag: Optional[int]
    last_significant_lag: Optional[int]

    minimum_p_value: float
    maximum_lm_statistic: float

    significant_density: float

    hypothesis_rejected: bool

    signal_to_noise: float
    signal_to_noise_capped: float


class ARCHStatisticsEngine:

    METADATA = {
        "module": 32,
        "name": "ARCH Effect Test",
        "version": "1.0.0",
        "alpha": 0.05,
        "max_lag": 20,
    }

    def analyze(self, result: ARCHResult) -> ARCHStatistics:

        significant_lags = int(result.significant_mask.sum())

        lags = result.lags[result.significant_mask]
        first_significant_lag = None if len(lags) == 0 else int(lags.min())
        last_significant_lag = None if len(lags) == 0 else int(lags.max())

        minimum_p_value = float(np.min(result.p_values))
        maximum_lm_statistic = float(np.max(result.lm_statistics))

        significant_density = significant_lags / result.max_lag if result.max_lag > 0 else 0.0

        hypothesis_rejected = significant_lags > 0

        signal_to_noise_raw = float(
            -np.log10(max(minimum_p_value, np.finfo(float).tiny))
        )
        signal_to_noise = round(signal_to_noise_raw, 2)
        signal_to_noise_capped = round(min(signal_to_noise_raw, 20.0), 2)

        return ARCHStatistics(
            sample_size=result.sample_size,
            alpha=result.alpha,
            max_lag=result.max_lag,
            significant_lags=significant_lags,
            first_significant_lag=first_significant_lag,
            last_significant_lag=last_significant_lag,
            minimum_p_value=minimum_p_value,
            maximum_lm_statistic=maximum_lm_statistic,
            significant_density=significant_density,
            hypothesis_rejected=hypothesis_rejected,
            signal_to_noise=signal_to_noise,
            signal_to_noise_capped=signal_to_noise_capped,
        )

    def summary_dataframe(self, stats: ARCHStatistics) -> pd.DataFrame:
        return pd.DataFrame([asdict(stats)])

    def summary_dict(self, stats: ARCHStatistics) -> dict:
        return asdict(stats)

    @staticmethod
    def cross_dataset_metrics(all_stats: list) -> pd.DataFrame:
        """Generate cross-dataset comparison metrics."""
        rows = []
        for name, stats in all_stats:
            rows.append({
                "dataset": name,
                "sample_size": stats.sample_size,
                "maximum_lm_statistic": round(stats.maximum_lm_statistic, 2),
                "minimum_p_value": round(stats.minimum_p_value, 6),
                "signal_to_noise": stats.signal_to_noise,
                "signal_to_noise_capped": stats.signal_to_noise_capped,
                "significant_lags": stats.significant_lags,
                "significant_density": round(stats.significant_density, 4),
                "hypothesis_rejected": stats.hypothesis_rejected,
            })
        return pd.DataFrame(rows)
