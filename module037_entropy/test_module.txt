import numpy as np

def test_gaussian():
    """Test 1: IID Gaussian returns → high entropy (> 0.90)."""
    print("\n" + "=" * 60)
    print("TEST 1: IID Gaussian Returns")
    print("=" * 60)

    engine = EntropyEngine(bins="fd")
    stats_engine = EntropyStatisticsEngine()

    np.random.seed(42)
    n = 2000
    returns = np.random.normal(0, 1, n)
    result = engine.compute(returns, series_type="log_returns")
    stats = stats_engine.analyze(result)

    assert result.normalized_entropy > 0.85, (
        f"Gaussian returns should have high entropy, got {result.normalized_entropy:.4f}"
    )

    print(f"  Entropy: {result.entropy:.4f} bits")
    print(f"  Normalized: {result.normalized_entropy:.4f}")
    print(f"  Bins: {len(result.probabilities)}")
    print(f"  Regime: {stats.regime}")
    print("  ✓ Gaussian test passed")
    return True


def test_constant():

    engine = EntropyEngine(bins="fd")

    return True


def test_binary():
    """Test 3: Binary series → entropy ≈ 1 bit."""
    print("\n" + "=" * 60)
    print("TEST 3: Binary Series")
    print("=" * 60)

    engine = EntropyEngine(bins="fd")
    stats_engine = EntropyStatisticsEngine()

    np.random.seed(42)
    n = 2000
    binary = np.random.choice([0, 1], size=n)
    result = engine.compute(binary, series_type="binary")
    stats = stats_engine.analyze(result)

    assert 0.8 < result.entropy < 1.2, (
        f"Binary series should have ~1 bit entropy, got {result.entropy:.4f}"
    )

    print(f"  Entropy: {result.entropy:.4f} bits")
    print(f"  Normalized: {result.normalized_entropy:.4f}")
    print(f"  Regime: {stats.regime}")
    print("  ✓ Binary test passed")
    return True

def test_pipeline():
    """Test 6: Full pipeline."""
    print("\n" + "=" * 60)
    print("TEST 6: Full Pipeline")
    print("=" * 60)

    n = 500
    np.random.seed(42)
    close = np.cumsum(np.random.normal(0, 1, n)) + 100
    close = np.maximum(close, 1)

    dates = pd.date_range(start="2020-01-01", periods=n, freq="15min")
    df = pd.DataFrame({
        "timestamp": dates,
        "open": close * 0.999,
        "high": close * 1.002,
        "low": close * 0.998,
        "close": close,
        "volume": np.ones(n) * 100,
    })

    research = EntropyResearch()
    result = research.run_dataframe(df, dataset_id="pipeline_test")

    assert result.status == "completed"
    assert "entropy_statistics" in result.artifacts
    assert "entropy_results" in result.artifacts
    assert "entropy_report" in result.artifacts
    assert result.statistics.get("validation_passed") == True

    print(f"  Entropy: {result.statistics.get('entropy'):.4f} bits")
    print(f"  Normalized: {result.statistics.get('normalized_entropy'):.4f}")
    print("  ✓ Full pipeline test passed")
    return True
