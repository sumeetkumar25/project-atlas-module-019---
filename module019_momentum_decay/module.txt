# atlas/modules/module019_momentum_decay/module.py
"""
Module 019
Momentum Decay

All 6 steps implemented.
"""

from typing import Optional

import pandas as pd

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from atlas.core.momentum import (
    ImpulseDetector,
    ImpulseThreshold,
    MomentumDecay,
    MomentumStatistics,
)

from .config import *
from .protocol import PROTOCOL


class MomentumDecayResearch(BaseResearchModule):

    PROTOCOL = PROTOCOL

    def __init__(
        self,
        registry: Optional[AtlasRegistry] = None,
    ):

        super().__init__(
            module_name=MODULE_NAME,
            module_version=MODULE_VERSION,
        )

        self.registry = registry or AtlasRegistry()

        if not self.registry.get_module(MODULE_ID):
            self.registry.register_module(PROTOCOL)

        self.impulse_df = None
        self.events = None
        self.decay_df = None
        self.statistics = None

    def _register_pipeline_steps(self):

        self.pipeline_steps = [

            ("impulse", self._detect_impulse),

            ("threshold", self._apply_threshold),

            ("decay", self._compute_decay),

            ("statistics", self._compute_statistics),

            ("distribution", self._compute_distribution),

            ("export", self._export_results),

        ]

    # ---- Step 1: Impulse Detection ----
    def _detect_impulse(self):

        self.logger.info("Step 1: Detecting impulses")
        self.logger.info(f"  Lookback: {LOOKBACK} bars")

        self.impulse_df = ImpulseDetector.detect(
            self.df["close"],
            lookback=LOOKBACK,
        )

        # Add timestamp for export
        self.impulse_df["timestamp"] = self.df["timestamp"]

        # Store full series
        self.result.artifacts["impulse_series"] = self.impulse_df.copy()

        up_count = int((self.impulse_df["direction"] == "UP").sum())
        down_count = int((self.impulse_df["direction"] == "DOWN").sum())

        self.logger.metric("Total observations", f"{len(self.impulse_df):,}")
        self.logger.metric("Up impulses", f"{up_count:,}")
        self.logger.metric("Down impulses", f"{down_count:,}")

        self.logger.success("Impulse detection complete")

        return {
            "impulse_status": "complete",
            "impulse_observations": len(self.impulse_df),
            "up_impulses": up_count,
            "down_impulses": down_count,
        }

    # ---- Step 2: Threshold Detection ----
    def _apply_threshold(self):

        self.logger.info("Step 2: Applying threshold")
        self.logger.info(f"  Window: {THRESHOLD_WINDOW}, Percentile: {THRESHOLD_PERCENTILE}")

        threshold = ImpulseThreshold.rolling_threshold(
            self.impulse_df["impulse_change"],
            window=THRESHOLD_WINDOW,
            percentile=THRESHOLD_PERCENTILE,
        )

        self.impulse_df["threshold"] = threshold

        self.events = ImpulseThreshold.detect(
            self.impulse_df["impulse_change"],
            threshold,
        )

        self.impulse_df["is_significant"] = self.events

        event_count = int(self.events.sum())
        event_pct = event_count / len(self.events) * 100 if len(self.events) > 0 else 0

        self.logger.metric("Significant events", f"{event_count:,}")
        self.logger.metric("Event frequency", f"{event_pct:.2f}%")

        # Export only significant events
        significant = self.impulse_df[self.impulse_df["is_significant"]].copy()
        self.result.artifacts["impulse_events"] = significant

        self.logger.success("Threshold detection complete")

        return {
            "threshold_status": "complete",
            "significant_events": event_count,
            "event_frequency": round(event_pct, 2),
        }

    # ---- Step 3: Momentum Decay ----
    def _compute_decay(self):

        self.logger.info("Step 3: Computing momentum decay")

        decay = MomentumDecay.compute(self.impulse_df["impulse_change"])
        self.impulse_df["decay"] = decay

        # Decay for significant events only
        significant = self.impulse_df[self.impulse_df["is_significant"]].copy()
        self.decay_df = significant

        self.result.artifacts["momentum_decay"] = self.decay_df.copy()

        if len(significant) > 0:
            mean_decay = float(significant["decay"].mean())
            std_decay = float(significant["decay"].std())
        else:
            mean_decay = 0.0
            std_decay = 0.0

        self.logger.metric("Decay observations", f"{len(significant):,}")
        self.logger.metric("Mean decay", f"{mean_decay:.6f}")
        self.logger.metric("Std decay", f"{std_decay:.6f}")

        self.logger.success("Decay computation complete")

        return {
            "decay_status": "complete",
            "decay_observations": len(significant),
            "mean_decay_raw": mean_decay,
            "std_decay_raw": std_decay,
        }

    # ---- Step 4: Statistics ----
    def _compute_statistics(self):

        self.logger.info("Step 4: Computing statistics")

        if self.decay_df is None or len(self.decay_df) == 0:
            self.statistics = {
                "event_count": 0,
                "mean_impulse": 0, "median_impulse": 0, "std_impulse": 0,
                "mean_decay": 0, "median_decay": 0, "std_decay": 0,
                "max_decay": 0, "min_decay": 0,
                "p25_decay": 0, "p75_decay": 0, "p90_decay": 0, "p95_decay": 0,
            }
        else:
            impulse_vals = self.decay_df["impulse_change"]
            decay_vals = self.decay_df["decay"]

            self.statistics = {
                "event_count": len(self.decay_df),
                "mean_impulse": float(impulse_vals.mean()),
                "median_impulse": float(impulse_vals.median()),
                "std_impulse": float(impulse_vals.std()),
                "mean_decay": float(decay_vals.mean()),
                "median_decay": float(decay_vals.median()),
                "std_decay": float(decay_vals.std()),
                "max_decay": float(decay_vals.max()),
                "min_decay": float(decay_vals.min()),
                "p25_decay": float(decay_vals.quantile(0.25)),
                "p75_decay": float(decay_vals.quantile(0.75)),
                "p90_decay": float(decay_vals.quantile(0.90)),
                "p95_decay": float(decay_vals.quantile(0.95)),
            }

        self.result.artifacts["momentum_statistics"] = pd.DataFrame([self.statistics])

        for k, v in self.statistics.items():
            if isinstance(v, float):
                self.logger.metric(k, f"{v:.6f}")
            else:
                self.logger.metric(k, f"{v:,}")

        self.logger.success("Statistics complete")

        return {
            "statistics_status": "complete",
            **self.statistics,
        }

    # ---- Step 5: Distribution ----
    def _compute_distribution(self):

        self.logger.info("Step 5: Computing distribution")

        if self.decay_df is None or len(self.decay_df) == 0:
            distribution = pd.DataFrame(columns=["decay_range", "count", "percentage"])
        else:
            decay = self.decay_df["decay"]

            bins = [float("-inf"), -0.05, -0.02, -0.01, 0, 0.01, 0.02, 0.05, float("inf")]
            labels = [
                "< -5%", "-5% to -2%", "-2% to -1%", "-1% to 0%",
                "0% to 1%", "1% to 2%", "2% to 5%", "> 5%"
            ]

            decay_range = pd.cut(decay, bins=bins, labels=labels)
            distribution = (
                decay_range
                .value_counts()
                .sort_index()
                .rename_axis("decay_range")
                .reset_index(name="count")
            )

            total = distribution["count"].sum()
            if total > 0:
                distribution["percentage"] = (
                    distribution["count"] / total * 100
                ).round(1)
            else:
                distribution["percentage"] = 0.0

        self.result.artifacts["momentum_distribution"] = distribution

        self.logger.metric("Distribution rows", f"{len(distribution)}")

        for _, row in distribution.iterrows():
            self.logger.info(
                f"  {row['decay_range']}: {int(row['count'])} ({row['percentage']:.1f}%)"
            )

        self.logger.success("Distribution complete")

        return {"distribution_status": "complete"}

    # ---- Step 6: Export ----
    def _export_results(self):

        self.logger.info("Step 6: Export")

        self.logger.metric("Artifacts", f"{len(self.result.artifacts)}")

        for name, artifact in self.result.artifacts.items():
            if hasattr(artifact, "__len__"):
                self.logger.info(f"  {name}: {len(artifact)} rows")

        self.logger.success("Export complete")

        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
        }

    def compute(self):

        self.logger.section("Momentum Decay Pipeline")
        self.logger.info("Executing Steps 1-6")

        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 019 complete")

        return results

    def run(self, filepath: str, output_dir=None):
        return super().run(filepath, output_dir)

    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        return super().run_dataframe(df, dataset_id, output_dir)
