"""

from pathlib import Path

from atlas.modules.module019_momentum_decay.module import (
    MomentumDecayResearch,
)


DATASETS = {

    "BTC_15m": Path(
        "data/raw/BTCUSD_15m.csv"
    ),

    "BTC_1h": Path(
        "data/raw/BTCUSD_1h.csv"
    ),

    "BTC_4h": Path(
        "data/raw/BTCUSD_4h.csv"
    ),

}


def validate_dataset(name, path):

    print()
    print("=" * 70)
    print(name)
    print("=" * 70)

    module = MomentumDecayResearch()

    result = module.run(

        filepath=str(path),

        output_dir=f"outputs/module019/{name}",

    )

    assert result.status == "completed"

    stats = result.statistics

    print()

    print(f"Events      : {stats['event_count']:,}")

    print(f"Frequency   : {stats['event_frequency']:.4f}%")

    print(f"Mean Impulse: {stats['mean_impulse']:.4f}")

    print(f"Mean Decay  : {stats['mean_decay']:.4f}")

    print(f"Std Decay   : {stats['std_decay']:.4f}")

    print()

    print("✓ Validation Passed")

    return result


def main():

    print()
    print("=" * 80)
    print("PROJECT ATLAS")
    print("MODULE 019 PRODUCTION VALIDATION")
    print("=" * 80)

    results = {}

    for name, path in DATASETS.items():

        results[name] = validate_dataset(
            name,
            path,
        )

    print()
    print("=" * 80)
    print("CROSS DATASET SUMMARY")
    print("=" * 80)

    for name, result in results.items():

        s = result.statistics

        print()

        print(name)

        print(f"Events     : {s['event_count']:,}")

        print(f"Frequency  : {s['event_frequency']:.4f}%")

        print(f"MeanDecay  : {s['mean_decay']:.4f}")

        print(f"StdDecay   : {s['std_decay']:.4f}")

    print()
    print("=" * 80)
    print("✓ MODULE 019 VALIDATION COMPLETE")
    print("=" * 80)


if __name__ == "__main__":

    main()
