def test_independent():
    """Test 1: Independent Gaussian series → MI near 0."""
    print("\n" + "=" * 60)
    print("TEST 1: Independent Gaussian Series")
    print("=" * 60)

    engine = MutualInformationEngine(bins="fd", lag=1)
    stats_engine = MutualInformationStatisticsEngine()

    np.random.seed(42)
    n = 2000
    x = np.random.normal(0, 1, n)
    y = np.random.normal(0, 1, n)
    result = engine.compute(x, y, series_type="independent_gaussian")
    stats = stats_engine.analyze(result)

    assert result.normalized_mutual_information < 0.10, (
        f"Independent series should have low NMI, got {result.normalized_mutual_information:.6f}"
    )
    assert stats.regime in ["Independent", "Weak Dependence"]

    print(f"  MI: {result.mutual_information:.6f} bits")
    print(f"  NMI: {result.normalized_mutual_information:.6f}")
    print(f"  Regime: {stats.regime}")
    print("  ✓ Independent test passed")
    return True

def test_pipeline():
    """Test 6: Full pipeline."""
    print("\n" + "=" * 60)
    print("TEST 6: Full Pipeline")
    print("=" * 60)

    n = 500
    np.random.seed(42)
    close = np.cumsum(np.random.normal(0, 1, n)) + 100
    close = np.maximum(close, 1)

    dates = pd.date_range(start="2020-01-01", periods=n, freq="15min")
    df = pd.DataFrame({
        "timestamp": dates,
        "open": close * 0.999,
        "high": close * 1.002,
        "low": close * 0.998,
        "close": close,
        "volume": np.ones(n) * 100,
    })

    research = MutualInformationResearch()
    result = research.run_dataframe(df, dataset_id="pipeline_test")

    assert result.status == "completed"
    assert result.statistics.get("lag") == 1
    assert "mi_statistics" in result.artifacts
    assert "mi_data" in result.artifacts
    assert "mi_report" in result.artifacts
    assert result.statistics.get("validation_passed") == True

    print(f"  MI: {result.statistics.get('mutual_information'):.6f}")
    print(f"  NMI: {result.statistics.get('normalized_mi'):.6f}")
    print("  ✓ Full pipeline test passed")
    return True
